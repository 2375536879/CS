-- III. 流调数据库 (Epidemiological Survey Database)

-- Table 1: survey_participants (流调参与者)
CREATE TABLE `survey_participants` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `patient_id` INT COMMENT 'FK to patients.id, nullable',
  `survey_identifier` VARCHAR(100) COMMENT '调查编号',
  `survey_date` DATE COMMENT '调查时间',
  `name` VARCHAR(255) COMMENT '姓名',
  `gender` ENUM('男', '女', '其他') COMMENT '性别',
  `age_at_survey` INT COMMENT '年龄',
  `residence_type` ENUM('城市', '农村') COMMENT '居住地类型',
  `residence_duration_years` INT COMMENT '居住时长（年）',
  `home_address_survey` TEXT COMMENT '家庭住址',
  `height_cm_survey` DECIMAL(5,2) COMMENT '身高',
  `weight_kg_survey` DECIMAL(5,2) COMMENT '体重',
  `has_previous_allergy_history` TINYINT(1) COMMENT '既往过敏史：有/无',
  `previous_allergic_rhinitis` TINYINT(1) COMMENT '既往:过敏性鼻炎',
  `previous_bronchial_asthma` TINYINT(1) COMMENT '既往:支气管哮喘',
  `previous_atopic_dermatitis` TINYINT(1) COMMENT '既往:特应性皮炎',
  `previous_allergic_conjunctivitis` TINYINT(1) COMMENT '既往:过敏性结膜炎',
  `previous_urticaria` TINYINT(1) COMMENT '既往:荨麻疹',
  `current_survey_diag_ar` TINYINT(1) COMMENT '本次调查诊断:过敏性鼻炎',
  `current_survey_diag_ba` TINYINT(1) COMMENT '本次调查诊断:支气管哮喘',
  `current_survey_diag_ad` TINYINT(1) COMMENT '本次调查诊断:特应性皮炎',
  `current_survey_diag_ac` TINYINT(1) COMMENT '本次调查诊断:过敏性结膜炎',
  `current_survey_diag_urticaria` TINYINT(1) COMMENT '本次调查诊断:荨麻疹',
  `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
  `updated_at` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_survey_identifier` (`survey_identifier`),
  FOREIGN KEY (`patient_id`) REFERENCES `patients`(`id`) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='流调参与者';

-- Table 2: survey_investigators (流调员信息)
CREATE TABLE `survey_investigators` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `survey_participant_id` INT NOT NULL COMMENT 'FK to survey_participants.id',
  `investigator_name` VARCHAR(255) COMMENT '姓名',
  `investigator_title` ENUM('住院医师', '主治医师', '副主任医师', '主任医师') COMMENT '职称',
  `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
  `updated_at` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`survey_participant_id`) REFERENCES `survey_participants`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='流调员信息';

-- Table 3: home_environment_exposure (家庭环境暴露信息)
CREATE TABLE `home_environment_exposure` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `survey_participant_id` INT NOT NULL COMMENT 'FK to survey_participants.id',
  `housing_type` ENUM('平房', '楼房', '别墅', '其他') COMMENT '房屋类型',
  `building_material` ENUM('木质', '混凝土', '其他') COMMENT '建筑材料',
  `ventilation_frequency` ENUM('每日', '每周', '偶尔') COMMENT '通风频率',
  `ac_usage_season` ENUM('几乎不用', '夏季每日', '全年高频') COMMENT '空调使用季节',
  `ac_usage_frequency` ENUM('每日', '每周3-5天', '偶尔') COMMENT '空调使用频率',
  `ac_temp_setting` ENUM('<24℃', '24-26℃', '>26℃') COMMENT '空调温度设定',
  `ac_mode` ENUM('制冷', '制热', '通风', '除湿') COMMENT '空调模式',
  `ac_filter_clean_freq` ENUM('每月', '每季度', '每年', '从未清洁') COMMENT '空调滤网清洁频率',
  `heating_type` ENUM('集中供暖', '电暖器', '燃气壁挂炉', '其他') COMMENT '暖气类型',
  `heating_room_temp_range` ENUM('<18℃', '18-22℃', '>22℃') COMMENT '室温范围',
  `has_carpet` TINYINT(1) COMMENT '地毯：有/无',
  `has_plush_toys` TINYINT(1) COMMENT '毛绒玩具：有/无',
  `home_pm25_avg_concentration` DECIMAL(10,2) COMMENT 'PM2.5家庭室内年均浓度 μg/m³',
  `home_pollen_types_concentration` TEXT COMMENT '室内花粉种类及浓度',
  `home_formaldehyde_concentration` DECIMAL(10,2) COMMENT '室内甲醛浓度 mg/m³',
  `home_formaldehyde_detection_time` DATETIME COMMENT '甲醛检测时间',
  `home_dust_mite_concentration` VARCHAR(255) COMMENT '床垫/沙发尘螨浓度 个/g灰尘',
  `home_other_allergens` TEXT COMMENT '其他（霉菌、宠物皮屑、蟑螂过敏原等）',
  `exercise_freq_intensity` VARCHAR(255) COMMENT '运动频率与强度',
  `swimming_habit` TINYINT(1) COMMENT '游泳有/无',
  `sleep_quality_duration` VARCHAR(255) COMMENT '睡眠质量与时长',
  `has_insomnia` TINYINT(1) COMMENT '失眠：有/无',
  `has_circadian_rhythm_disorder` TINYINT(1) COMMENT '昼夜节律紊乱：有/无',
  `smoking_alcohol_e_cigarette_use` ENUM('主动', '被动暴露', '无') COMMENT '吸烟/饮酒/电子烟使用情况',
  `cleaning_frequency` ENUM('每日', '每周', '每月') COMMENT '清洁频率',
  `pet_cat` TINYINT(1) COMMENT '宠物饲养：猫',
  `pet_dog` TINYINT(1) COMMENT '宠物饲养：狗',
  `pet_bird` TINYINT(1) COMMENT '宠物饲养：鸟',
  `pet_other` VARCHAR(255) COMMENT '宠物饲养：其它',
  `pet_quantity` INT COMMENT '宠物数量',
  `cohabiting_smoker` TINYINT(1) COMMENT '吸烟者同居情况：有/无',
  `cooking_fuel_type` ENUM('天燃气', '煤气', '生物质燃料') COMMENT '烹饪燃料',
  `uses_air_purifier` TINYINT(1) COMMENT '空气净化器使用：是/否',
  `uses_vacuum_cleaner` TINYINT(1) COMMENT '吸尘器使用：是/否',
  `allergen_avoidance_execution` ENUM('优', '良', '差') COMMENT '过敏原回避措施执行情况',
  `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
  `updated_at` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`survey_participant_id`) REFERENCES `survey_participants`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='家庭环境暴露信息';

-- Table 4: work_study_environment_exposure (学习/工作环境信息)

  `pm25_exposure_annual_avg` DECIMAL(10,2) COMMENT 'PM2.5暴露水平年均值 μg/m³',
  `pollen_exposure_peak_concentration` VARCHAR(255) COMMENT '花粉暴露季节性峰值浓度 颗粒/m³',
  `pollen_types_work_study` TEXT COMMENT '花粉种类',
  `formaldehyde_new_decor_value` VARCHAR(255) COMMENT '新装修教室/办公室甲醛检测值',
  `dust_mite_exposure_carpet` TINYINT(1) COMMENT '尘螨暴露:地毯有/无',
  `dust_mite_exposure_fabric_furniture` TINYINT(1) COMMENT '布艺家具使用情况：有/无',
  `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
  `updated_at` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

  FOREIGN KEY (`survey_participant_id`) REFERENCES `survey_participants`(`id`) ON DELETE CASCADE
