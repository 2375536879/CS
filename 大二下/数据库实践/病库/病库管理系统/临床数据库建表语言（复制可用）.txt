-- I. 临床数据库 (Clinical Database)

-- Table 1: patients (患者基本信息)
CREATE TABLE `patients` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `hospital_patient_id` VARCHAR(255) NOT NULL COMMENT '就诊号',
  `name` VARCHAR(255) NOT NULL COMMENT '姓名',
  `gender` ENUM('男', '女', '其他') COMMENT '性别',
  `date_of_birth` DATE COMMENT '出生日期',
  `home_address` TEXT COMMENT '家庭住址',
  `birth_weight_kg` DECIMAL(5,3) COMMENT '出生体重 kg',
  `lifestyle_notes` TEXT COMMENT '生活方式备注',
  `positive_food_allergen_history` TEXT COMMENT '食物过敏原阳性史',
  `positive_inhaled_allergen_history` TEXT COMMENT '吸入过敏原阳性史',
  `allergic_disease_history` TEXT COMMENT '过敏性疾病史',
  `family_allergy_history_degree1` TEXT COMMENT '一代亲属过敏史',
  `family_allergic_disease_history_degree1` TEXT COMMENT '一代亲属过敏性疾病史',
  `family_allergy_history_degree2` TEXT COMMENT '二代亲属过敏史',
  `family_allergic_disease_history_degree2` TEXT COMMENT '二代亲属过敏性疾病史',
  `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
  `updated_at` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_hospital_patient_id` (`hospital_patient_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='患者基本信息';

-- Table 2: patient_contacts (患者联系人)
CREATE TABLE `patient_contacts` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `patient_id` INT NOT NULL COMMENT 'FK to patients.id',
  `contact_name` VARCHAR(255) COMMENT '姓名',
  `contact_phone` VARCHAR(50) COMMENT '联系方式',
  `relationship` VARCHAR(100) COMMENT '与患者关系',
  `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
  `updated_at` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`patient_id`) REFERENCES `patients`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='患者联系人';

-- Table 3: insurance_details (医保情况)
CREATE TABLE `insurance_details` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `patient_id` INT NOT NULL COMMENT 'FK to patients.id',
  `insurance_type` VARCHAR(255) COMMENT '医保类型',
  `insurance_number` VARCHAR(255) COMMENT '医保号',
  `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
  `updated_at` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`patient_id`) REFERENCES `patients`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='医保情况';

-- Table 4: clinical_visits (就诊记录)
CREATE TABLE `clinical_visits` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `patient_id` INT NOT NULL COMMENT 'FK to patients.id',
  `visit_date` DATETIME COMMENT '就诊时间',
  `height_cm` DECIMAL(5,2) COMMENT '身高 cm',
  `weight_kg` DECIMAL(5,2) COMMENT '体重 kg',
  `healthcare_provider_name` VARCHAR(255) COMMENT '医疗服务者姓名',
  `healthcare_provider_title` VARCHAR(255) COMMENT '医疗服务者职称',
  `institution_name` VARCHAR(255) COMMENT '机构名称',
  `institution_address` TEXT COMMENT '机构地址',
  `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
  `updated_at` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`patient_id`) REFERENCES `patients`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='就诊记录';
