一、创建用户表（user）
二、商品表（sh_goods）
三、购物车表（cart）
四、收货地址表（address）
五、订单与订单商品表（order + order_item）
六、商品评分表（goods_rating）
七、商品评论表（sh_goods_comment）

CREATE TABLE `user` (
    id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(20) NOT NULL UNIQUE COMMENT '用户名，可使用中文',
    phone CHAR(11) NOT NULL COMMENT '手机号码',
    gender ENUM('男', '女', '保密') DEFAULT '保密' COMMENT '性别',
    register_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '注册时间',
    member_level TINYINT UNSIGNED DEFAULT 0 COMMENT '会员等级，最高为100',
    CHECK (CHAR_LENGTH(username) BETWEEN 2 AND 20)
);


CREATE TABLE sh_goods (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL COMMENT '商品名称',
    price DECIMAL(10,2) NOT NULL COMMENT '价格',
    stock INT NOT NULL DEFAULT 0 COMMENT '库存'
);


CREATE TABLE cart (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    goods_id INT NOT NULL,
    quantity INT NOT NULL DEFAULT 1,
    added_time DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES `user`(id),
    FOREIGN KEY (goods_id) REFERENCES sh_goods(id),
    UNIQUE (user_id, goods_id)
);


CREATE TABLE address (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    receiver_name VARCHAR(50) NOT NULL,
    phone CHAR(11) NOT NULL,
    province VARCHAR(50),
    city VARCHAR(50),
    district VARCHAR(50),
    detail_address TEXT,
    is_default BOOLEAN DEFAULT FALSE,
    FOREIGN KEY (user_id) REFERENCES `user`(id)
);


CREATE TABLE `order` (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    address_id INT NOT NULL,
    total_price DECIMAL(10,2) NOT NULL,
    status ENUM('待付款', '已付款', '已发货', '已完成', '已取消') DEFAULT '待付款',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP,
    pay_time DATETIME NULL DEFAULT NULL,
    FOREIGN KEY (user_id) REFERENCES `user`(id),
    FOREIGN KEY (address_id) REFERENCES address(id)
);


CREATE TABLE order_item (
    id INT AUTO_INCREMENT PRIMARY KEY,
    order_id INT NOT NULL,
    goods_id INT NOT NULL,
    quantity INT NOT NULL,
    price DECIMAL(10,2) NOT NULL,
    FOREIGN KEY (order_id) REFERENCES `order`(id),
    FOREIGN KEY (goods_id) REFERENCES sh_goods(id)
);



CREATE TABLE goods_rating (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    goods_id INT NOT NULL,
    rating TINYINT UNSIGNED NOT NULL CHECK (rating BETWEEN 1 AND 5),
    comment TEXT,
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES `user`(id),
    FOREIGN KEY (goods_id) REFERENCES sh_goods(id),
    UNIQUE (user_id, goods_id)
);



CREATE TABLE sh_goods_comment (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    goods_id INT NOT NULL,
    content TEXT NOT NULL,
    is_valid BOOLEAN DEFAULT TRUE,
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES `user`(id),
    FOREIGN KEY (goods_id) REFERENCES sh_goods(id)
);

/*
 查询实践部分
1. 查询商品id等于8且有效的评论内容
sql
深色版本
SELECT * FROM sh_goods_comment
WHERE goods_id = 8 AND is_valid = TRUE;
2. 查询每个用户评论的商品数量
sql
深色版本
SELECT user_id, COUNT(DISTINCT goods_id) AS comment_count
FROM sh_goods_comment
WHERE is_valid = TRUE
GROUP BY user_id;
3. 查询最新发布的5条有效商品评论信息
sql
深色版本
SELECT * FROM sh_goods_comment
WHERE is_valid = TRUE
ORDER BY create_time DESC
LIMIT 5;
4. 查询评论过两种以上不同商品的用户id及对应的商品id
sql
深色版本
SELECT user_id, goods_id
FROM sh_goods_comment
WHERE is_valid = TRUE
AND user_id IN (
    SELECT user_id
    FROM sh_goods_comment
    WHERE is_valid = TRUE
    GROUP BY user_id
    HAVING COUNT(DISTINCT goods_id) >= 2
);
5. 查询没有任何评论信息的商品id和name
sql
深色版本
SELECT g.id, g.name
FROM sh_goods g
LEFT JOIN sh_goods_comment c ON g.id = c.goods_id AND c.is_valid = TRUE
WHERE c.id IS NULL;
6. 查询商品评分为5星的商品评论信息
sql
深色版本
SELECT gc.*
FROM goods_rating gr
JOIN sh_goods_comment gc ON gr.user_id = gc.user_id AND gr.goods_id = gc.goods_id
WHERE gr.rating = 5 AND gc.is_valid = TRUE;
 */
